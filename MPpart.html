async function submitMarketplaceItem() { const itemType = document.getElementById('mp-sell-type').value; const qtyInput = parseInt(document.getElementById('mp-sell-qty').value, 10); const priceInput = parseInt(document.getElementById('mp-sell-price').value, 10); if (isNaN(qtyInput) || qtyInput <= 0) { showToast("Please enter a valid quantity in kg."); return; } if (isNaN(priceInput) || priceInput <= 0) { showToast("Please enter a valid ask price."); return; } let maxAllowedPrice = 0; if (itemType === 'food') { maxAllowedPrice = Math.floor(qtyInput * 1.60); } else { maxAllowedPrice = qtyInput * 80; } if (priceInput > maxAllowedPrice) { showToast(`Price too high! Maximum allowed for ${qtyInput}kg ${itemType} is ${maxAllowedPrice} Rinth.`); return; } // 1. Haal de ingelogde gebruiker op uit Supabase Auth const { data: { user } } = await dbClient.auth.getUser(); if (!user) { showToast("You must be logged in to list items."); return; } // 2. Haal de username op uit de designers-tabel op basis van user.id const { data: designerData } = await dbClient .from('designers') .select('username') .eq('id', user.id) .single(); try { // 3. Bouw de juiste payload met designer_id en username const payload = { item_type: itemType, quantity: Number(qtyInput), price_rinth: Number(priceInput), designer_id: user.id, username: designerData ? designerData.username : null }; const { data, error } = await dbClient.from('marketplace').insert([payload]); if (error) { console.error("Supabase Error Details:", error); showToast("Failed: " + (error.message || "Database error")); return; } showToast("Item successfully listed on the marketplace!"); document.getElementById('mp-sell-qty').value = "1"; document.getElementById('mp-sell-price').value = "1"; fetchMarketplaceItems(); } catch (err) { console.error("Catch Error:", err); showToast("An error occurred while listing the item."); } }
